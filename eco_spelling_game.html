    let currentIndex = 0;

    function loadQuestion() {

        const item = quizData[currentIndex];

        document.getElementById("user-input").value = "";

        document.getElementById("feedback").textContent = "";

        document.getElementById("feedback").className = "feedback";

        document.getElementById("question-count").textContent = `Question ${currentIndex + 1} / ${quizData.length}`;

        

        const badge = document.getElementById("category-badge");

        if (item.isBonus) {

            badge.style.color = "#1b5e20";

        }


        const progressPercent = ((currentIndex) / quizData.length) * 100;

        document.getElementById("progress-bar").style.width = progressPercent + "%";


        // Speak word automatically after a small delay

        setTimeout(() => {

            speakCurrentWord();

        }, 400);

    }


    function speakCurrentWord() {

        const word = quizData[currentIndex].word;

        if ('speechSynthesis' in window) {

            window.speechSynthesis.cancel(); // stop previous speech

            const utterance = new SpeechSynthesisUtterance(word);

            utterance.lang = 'en-US';

            utterance.rate = 0.85; // slightly slower for standard dictation clarity

            window.speechSynthesis.speak(utterance);

        }

    }


    function checkAnswer(e) {

        e.preventDefault();

        const userInput = document.getElementById("user-input").value.trim().toLowerCase();

        const targetWord = quizData[currentIndex].word.toLowerCase();

        const feedbackEl = document.getElementById("feedback");


        if (!userInput) return;


        if (userInput === targetWord) {

            score += quizData[currentIndex].isBonus ? 15 : 10;

            document.getElementById("score").textContent = score;

            feedbackEl.textContent = "🎉 Excellent! Correct Spelling!";

            feedbackEl.className = "feedback correct";


            setTimeout(() => {

                currentIndex++;

                if (currentIndex < quizData.length) {

                    loadQuestion();

                } else {

                    showResults();

                }

            }, 1200);

        } else {

            // Provide helpful spelling hint

            feedbackEl.textContent = `❌ Oops! Correct spelling: "${targetWord}". Try typing it again!`;

            feedbackEl.className = "feedback incorrect";

            speakCurrentWord();

        }

    }


    function showResults() {

        document.getElementById("progress-bar").style.width = "100%";

        document.getElementById("game-screen").style.display = "none";

        document.getElementById("results-screen").style.display = "block";

        document.getElementById("final-score").textContent = `Final Eco Score: ${score} Points!`;

    }


    function restartGame() {
